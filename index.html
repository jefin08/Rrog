Create the Server

Caption: Create the dashboard logic

Contains the reactive() data filtering expression that recalculates automatically when inputs change.

R Script
server <- function(input, output) {
  
  filtered_data <- reactive({
    
    data <- students
    
    if (input$department != "All") {
      data <- data[data$Department == input$department, ]
    }
    
    if (input$semester != "All") {
      data <- data[data$Semester == as.numeric(input$semester), ]
    }
    
    data <- data[data$Mark >= input$min_mark, ]
    
    data
  })
  
  
  output$total_students <- renderText({
    nrow(filtered_data())
  })
  
  
  output$average_mark <- renderText({
    round(mean(filtered_data()$Mark), 2)
  })
  
  
  output$average_attendance <- renderText({
    round(mean(filtered_data()$Attendance), 2)
  })
  
  
  output$student_table <- renderDT({
    filtered_data()
  })
  
  
  output$marks_plot <- renderPlot({
    
    ggplot(
      filtered_data(),
      aes(x = Mark)
    ) +
      geom_histogram(
        bins = 10
      ) +
      labs(
        title = "Marks Distribution",
        x = "Mark",
        y = "Number of Students"
      )
  })
  
  
  output$department_plot <- renderPlot({
    
    dept_data <- filtered_data() %>%
      group_by(Department) %>%
      summarise(
        Average_Mark = mean(Mark)
      )
    
    ggplot(
      dept_data,
      aes(x = Department, y = Average_Mark)
    ) +
      geom_col() +
      labs(
        title = "Department-wise Average Marks",
        x = "Department",
        y = "Average Mark"
      )
  })
}

Create the Server

Caption: Create the dashboard logic
R Script
server <- function(input, output) {
  
  
  filtered_data <- reactive({
    
    mtcars[
      mtcars$cyl %in% input$cyl &
      mtcars$mpg >= input$min_mpg,
    ]
  })
  
  
  output$car_count <- renderText({
    nrow(filtered_data())
  })
  
  
  output$average_mpg <- renderText({
    round(mean(filtered_data()$mpg), 2)
  })
  
  
  output$average_hp <- renderText({
    round(mean(filtered_data()$hp), 2)
  })
  
  
  output$car_table <- renderDT({
    filtered_data()
  })
  
  
  output$mpg_plot <- renderPlot({
    
    ggplot(
      filtered_data(),
      aes(x = mpg)
    ) +
      geom_histogram(
        bins = 8
      ) +
      labs(
        title = "MPG Distribution",
        x = "MPG",
        y = "Number of Cars"
      )
  })
  
  
  output$cylinder_plot <- renderPlot({
    
    cylinder_data <- filtered_data() %>%
      group_by(cyl) %>%
      summarise(
        Average_MPG = mean(mpg)
      )
    
    ggplot(
      cylinder_data,
      aes(x = factor(cyl), y = Average_MPG)
    ) +
      geom_col() +
      labs(
        title = "Average MPG by Cylinder",
        x = "Cylinders",
        y = "Average MPG"
      )
  })
}

💡 The Most Important Concept: reactive()

Whenever the user changes a filter, the filtered data is recalculated automatically.

Then the KPI text, table, and graphs call filtered_data() with parentheses so they re-render immediately.
